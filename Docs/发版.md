git merge dev --no-ff -m "发版: v0.3.3 | 合并自 DEV 提交: $(git rev-parse dev)"

# 3. main 上打标签（唯一正确的发版标签）
git tag v0.3.3
git push origin main --tags

# 删除本地所有标签
git tag -d $(git tag)
# 删除远程所有标签
git push origin --delete $(git tag)
