# 1. squash 合并 dev（会有冲突，因为无公共祖先）
git merge --squash dev --allow-unrelated-histories
# 2. 全部取 dev 版本（解决所有冲突）
git checkout --theirs -- .
git add -A
# 3. 提交（PowerShell 写法）
$devHash = git rev-parse dev
$version = "v0.1.13"
git commit -m "发版 $version | 源DEV提交: $devHash"
# 4. 标签 + 推送
git tag $version
git push origin main --tags
